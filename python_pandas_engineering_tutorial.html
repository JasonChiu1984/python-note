from __future__ import annotations

import json
from pathlib import Path
from typing import Any

import pandas as pd

REQUIRED_COLUMNS = {"order_id", "customer_id", "country", "created_at", "amount"}


def load_orders(csv_path: Path) -> pd.DataFrame:
    orders = pd.read_csv(
        csv_path,
        keep_default_na=False,
        # 只把契約欄位的空欄當缺值。NA 國別碼與 NULL 識別碼可合法存在。
        na_values={column: [""] for column in REQUIRED_COLUMNS},
        dtype={
            "order_id": "string",
            "customer_id": "string",
            "country": "string",
            "amount": "float64",
        },
    )
    missing = REQUIRED_COLUMNS - set(orders.columns)
    if missing:
        raise ValueError(f"missing columns: {sorted(missing)}")

    orders["created_at"] = pd.to_datetime(orders["created_at"], utc=True, errors="raise")
    return orders


def build_report(orders: pd.DataFrame) -> dict[str, Any]:
    amount_missing_ratio = round(float(orders["amount"].isna().mean()), 4) if len(orders) else None
    clean = orders.loc[orders["amount"].notna()].copy()
    if clean["amount"].isin([float("inf"), float("-inf")]).any():
        raise ValueError("amount must be finite")
    clean["month"] = clean["created_at"].dt.strftime("%Y-%m")

    revenue = (
        clean.groupby(["country", "month"], dropna=False)
        .agg(revenue=("amount", "sum"), orders=("order_id", "nunique"))
        .reset_index()
        .sort_values(["country", "month"])
    )

    return {
        "quality": {
            "rows": int(len(orders)),
            "amount_missing_ratio": amount_missing_ratio,
            "countries": sorted(clean["country"].dropna().unique().tolist()),
        },
        "revenue": revenue.astype(object).where(revenue.notna(), None).to_dict(orient="records"),
    }


def write_report(csv_path: Path, output_path: Path) -> Path:
    report = build_report(load_orders(csv_path))
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(json.dumps(report, ensure_ascii=False, indent=2, allow_nan=False), encoding="utf-8")
    return output_path

def test_build_report_quality_and_revenue(tmp_path):
    csv_path = tmp_path / "orders.csv"
    csv_path.write_text(
        "order_id,customer_id,country,created_at,amount\n"
        "o-001,c-001,TW,2026-05-01T10:20:00+08:00,1200\n"
        "o-002,c-002,JP,2026-05-02T09:30:00+09:00,\n",

可對帳的 DataFrame join：筆數不變也要保住能量

將能耗明細連到設備主檔之前，先定義資料粒度：左邊一列是一個讀值，右邊每個設備只能對應一筆主檔。Pandas merge 可以把兩側空鍵相互配對，不能照搬一般 SQL NULL 的假設。空鍵應事先隔離；缺主檔的有效鍵應保留在帳上並列為一個類別。

一、輸入型別、缺值與拷貝語意

解析是一份契約

範例以 dtype= 將設備 ID 讀為 string，不讓 001 變成整數 1；Float64 則保留電力缺值。以 keep_default_na=False 及欄位專用 na_values 定義空白規則，不讓其他欄位中可能合法的字串 NA 自動成為缺值。convert_dtypes() 可協助探索，不替代業務契約與欄位驗證。離線實驗先用 csv.reader 核對原始欄名與每筆欄數，再交給 Pandas。第一筆多一欄時，Pandas 可能把前置欄推成索引，欄名仍全部正確。整行為空可略過；kw 欄空白且狀態為 MISSING 則保留為缺值。大型檔案需改用對應的串流驗證策略，不應直接照搬此雙次解析的記憶體實驗。參見 Pandas 隱含索引說明。

缺值不是零

isna() 判斷缺值，不用 == pd.NA。值為 0 代表設備在該段時間的平均功率為零，缺值是未知；若兩者都填 0，數值總和雖可能不變，覆蓋率和推論卻會改變。規則先檢查讀值和品質旗標一致，再拒絕負值與無限大。

標籤會自動對齊

兩個 Series 相加通常先對齊 index，不是依實體位置配對。位置是資料契約時要先檢查順序和長度；標籤是契約時要檢查 index 唯一性。不要為了消除缺值就先 reset_index(drop=True)。

更新要明確

用 df.loc[mask, "kw"] = 0 表達對哪一個 DataFrame 寫入。Pandas 3.0 起 Copy-on-Write 是唯一模式，鏈式指派不會回寫原表。即使在舊版本觀察到鏈式指派可動，也不應依賴。不能僅由有沒有呼叫 copy() 推論峰值記憶體。

assert df["device_id"].dtype.name == "string"
assert not df.duplicated(["device_id", "event_at"]).any()
assert df["kw"].notna().eq(df["status"].eq("GOOD")).all()

上面是可放入單元測試的不變量，不是正式驗證器；正式入口應主動拋出例外，不依賴 Python 的 assert。

反例：國別代碼 NA 不是缺值。同一欄可以同時有納米比亞代碼 NA 與真正空白。僅固定字串型別不會改掉 CSV 的預設缺值標記。若欄位契約只承認空白是缺值，須明確指定：

import io
import pandas as pd

csv = "country,amount\nNA,20\n,5\nTW,\n"
df = pd.read_csv(io.StringIO(csv), dtype={"country": "string", "amount": "Float64"},
                 keep_default_na=False, na_values={"country": [""], "amount": [""]})
assert df["country"].eq("NA").fillna(False).sum() == 1
assert df["country"].isna().sum() == 1
assert df["amount"].sum() == 25  # 只比對總額會漏掉國別錯配

此頁可下載的訂單 ETL 也採用這項欄位政策，保留合法的國別碼和識別碼；測試會分開核對國別分組與總額。依來源規格決定其他代碼與空白字串如何處理。pandas read_csv、聯合國國別資料。

二、關聯運算：基數決定列數

遙測是事實表，資產資料是每設備一列的維度表。對單一鍵 k，若兩側各有 Lₖ、Rₖ 列，內連接會產生 Lₖ × Rₖ 列：3 筆事件撞到 2 列同設備資料會得到 6 列，不是新增了 3 筆觀測。若不先驗基數，事後 groupby().sum() 可能把同一筆 kWh 算兩次。

out = events.merge(assets, on="device_id", how="left",
                   validate="many_to_one", indicator=True)

validate= 驗證維度端唯一性，indicator= 才能辨認哪個事件找不到對應。Pandas 的 merge 允許兩側空鍵互相匹配，不等同一般 SQL NULL 等值連接；這裡直接拒絕空鍵，不用 drop_duplicates() 隱藏衝突。若維度會變更，改用有效期間或可得時間，不以最新快照回填歷史。

三、事件時間、可得時間與窗口

計算窗口與失敗邊界

每列代表固定 15 分鐘區間的平均 kW，kWh = kW × 0.25 h。如果是不定期瞬間取樣，必須另訂積分規則、最大間隔及缺值政策；不能直接套本式。此例用 UTC 與 [start, end) 左閉右開；先驗對齊與唯一鍵。若由本地時間匯入，無偏移量資料必須有明確時區，遇 DST 的重複／不存在時段要拒絕或依規則處理；不要把無時區當作 UTC。

分組的列數、有效值與時段分母

size() 算收到幾列，count() 算幾列有值，sum(min_count=1) 避免全缺值輸出零。每小時預期 4 槽，不以收到的筆數當覆蓋率分母。整小時完全沒列時，實驗輸出不產生該群組；正式系統要先建立「設備 × 應有時段」格網，才能報告沒有任何觀測的時間。

未來資料洩漏

校正值實際可用時間是 available_at，不一定是校正作業時間。merge_asof 先依時間排序，再依設備、向後、有限容許間隔對齊；這裡禁止等時匹配，避免無法判斷同時寫入的先後。離線模型需使用時間切分，任何補值、標準化或選特徵的參數都只在訓練期估計。若有晚到資料，回測應以當時可得的快照重跑。

滾動窗不是固定列數

rolling(4) 是四列，對缺測不等於一小時。用 rolling("1h", closed="left", min_periods=...) 時須指定窗口規則、事件時鐘和設備內排序，並在預測時刻只用之前窗口。即使窗口排除了目前列，仍需驗證每列輸入當時可取得。

確定性核對：設備 001 收到 3 列，有效 2 列，其中 0 kW 是有效值；已知電量為 (0 + 4) × 0.25 = 1.0 kWh，覆蓋率 2 / 4 = 50%，完整小時電量仍未知。00:15 的事件只能用 23:55 已可得的校正值 0.2，不會使用 00:16 才出現的 3.0。

延伸學習：實作、現場案例與自我檢查

本頁說明 pandas 的工程化做法。本節聚焦兩個最容易悄悄出錯的操作：merge 時點位表有重複鍵導致資料列被複製，以及把不同時間戳的讀值與設定點對齊。

可實跑範例：merge 驗證與 merge_asof 時間對齊

validate='many_to_one' 讓 pandas 在右表鍵重複時直接報錯，而不是默默產生多出來的列；merge_asof 以 direction='backward' 讓每筆讀值配上當時已生效的設定點。範例需要 pandas 3.x。

下載範例 example.py

"""pandas 工程化：merge 驗證關聯、對齊時間戳（merge_asof）、檢查合併前後筆數（需 pandas 3.x）。"""
import pandas as pd

readings = pd.DataFrame({
    "ts": pd.to_datetime(["2026-11-20 08:00:03", "2026-11-20 08:00:07", "2026-11-20 08:00:12",
                          "2026-11-20 08:00:03", "2026-11-20 08:00:09"]),
    "point": ["SAT", "SAT", "SAT", "KW", "KW"],
    "value": [22.4, 22.5, 22.7, 128.0, 131.5],
})
points = pd.DataFrame({"point": ["SAT", "KW", "KW"], "unit": ["degC", "kW", "kW"],
                       "ahu": ["AHU-01", "AHU-01", "AHU-02"]})          # 點位表中 KW 重複了
setpoints = pd.DataFrame({"ts": pd.to_datetime(["2026-11-20 08:00:00", "2026-11-20 08:00:10"]),
                          "sat_sp": [22.0, 23.0]})

if __name__ == "__main__":
    try:
        readings.merge(points, on="point", how="left", validate="many_to_one")
    except pd.errors.MergeError as e:
        print("merge 驗證失敗：", str(e).splitlines()[0])
    dup = points[points.duplicated("point", keep=False)]
    print("點位表重複鍵：", dup.to_dict("records"))
    clean = points.drop_duplicates("point", keep="first")
    merged = readings.merge(clean, on="point", how="left", validate="many_to_one")
    print(f"合併前 {len(readings)} 筆、合併後 {len(merged)} 筆（必須相同）")
    sat = readings[readings["point"] == "SAT"].sort_values("ts")
    aligned = pd.merge_asof(sat, setpoints.sort_values("ts"), on="ts", direction="backward")
    aligned["error"] = (aligned["value"] - aligned["sat_sp"]).round(2)
    print(aligned[["ts", "value", "sat_sp", "error"]].to_string(index=False))

工程現場案例：合併點位表後，用電量多算了一倍

某報表把電表讀值與點位表合併以取得單位與所屬空調箱。點位表中 KW 這個點位不小心出現兩次（同時對應 AHU-01 與 AHU-02），一般的 merge 會把每筆 KW 讀值複製成兩列，加總後用電量多了一倍，而且不會有任何錯誤訊息。

加上 validate 參數後，重複鍵會立即被發現；合併後再檢查筆數與合併前相同，是另一道防線。設定點對齊則不能用一般 merge（時間戳不會剛好相等），要用 merge_asof 找出每筆讀值當下有效的設定點。

  1. 每次 merge 都加上 validate，明確宣告預期的關聯（one_to_one、many_to_one 等）。
  2. 合併前後檢查筆數，左合併時兩者應相同。
  3. 時間序列對齊用 merge_asof，兩邊都先依時間排序，並視需要設定 tolerance 上限。
  4. 點位表本身在載入時就檢查鍵的唯一性。

練習題與自我檢查

  1. 不加 validate 直接 merge，合併後會有幾筆？

    看解答

    SAT 3 筆各配 1 列，KW 2 筆各配 2 列，共 3 + 4 = 7 筆，比原本多 2 筆。若對 value 加總，KW 的用電量會被重複計算。

  2. merge_asof 的 direction='backward' 是什麼意思？為什麼適合設定點？

    看解答

    每筆讀值配上時間早於或等於它的最近一筆設定點，也就是「當時已生效的設定」。用 forward 會配到未來才生效的設定，邏輯上錯誤。

  3. 若某筆讀值比所有設定點都早（例如 07:59），merge_asof 會給什麼？

    看解答

    sat_sp 為缺值。這代表沒有可用的設定點，應該在報表中標示，而不是用其他值填補。

  4. drop_duplicates('point', keep='first') 真的是正確的處理嗎？

    看解答

    不一定。範例為了示範後續步驟直接保留第一筆，但實際上 KW 對應兩台空調箱代表點位命名有問題，正確做法是修正點位表（例如改成 AHU1_KW、AHU2_KW），而不是任意丟掉一筆。
